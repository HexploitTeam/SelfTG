# SelfTG runtime dependencies.
#
# Replit installs this file with pip/upm once poetry.lock is absent (see
# .replit). Everything below is pinned exactly so a Rebuild pulls the same
# versions instead of whatever is newest that day.
#
# --- Telegram transport -----------------------------------------------------
# Upstream Telethon. The Replit fork (Telethon-repl, frozen at 1.33.1) is gone:
# the 5222 port it forced is pinned in Com/core/net.py instead, so this line can
# move forward without losing the Replit transport.
# Never install Telethon-repl next to this line: both provide `telethon`.
telethon==1.44.0

# --- HTTP (used by ~15 command modules) -------------------------------------
# 3.8.4 (the old pin) has open CVE-2023-47627 and CVE-2024-23334.
aiohttp==3.10.11

# --- Command modules --------------------------------------------------------
psutil==5.9.8            # !sysinfo, !sysmon, !uptime
pytz==2024.1             # timezone-aware timestamps in log/stats commands
Pygments==2.18.0         # !syntax, !prettify
Pillow==10.4.0           # image commands; also required by qrcode/moviepy
qrcode==7.4.2            # !qr
PyPDF2==3.0.1            # !pdfinfo, !convertpdf
mutagen==1.47.0          # !audiometa
moviepy==2.2.1           # !videometa. 1.x was removed from PyPI; 2.x has no `moviepy.editor`
                         # (Com/mediatools.py imports `moviepy` and loads it lazily)
gTTS==2.5.1              # !tts
Faker==24.14.0           # !mock
beautifulsoup4==4.12.3   # !prettify --type html (was imported but never pinned)
speedtest-cli==2.1.3     # !speedtest (was guarded as "not installed" on every run)
base58==2.1.1            # !encode base58
dnspython==2.6.1         # !dnsprobe, reverse DNS in !ip
Flask==3.0.3             # dashboard.py
requests==2.32.3         # gTTS transport
# Removed in v2: googletrans==4.0.2 and httpx==0.28.1.
# !translate now calls the same free Google endpoint directly over aiohttp
# (Com/translate.py) with MyMemory as fallback. googletrans' `translate` is a
# coroutine in 4.x, so the old `to_thread()` wrapper could never have returned a
# result, and it was the only reason httpx was installed.

# --- Optional acceleration --------------------------------------------------

# correct but slow, and Replit's CPU budget is small. Safe to remove: Telethon
# falls back automatically if it is missing or fails to load.
