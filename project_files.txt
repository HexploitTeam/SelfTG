# Com/blockuser.py
from telethon import events
from telethon import functions

__HELP__ = "!block @username / reply → Block user\n!unblock @username / reply → Unblock user"
__CATEGORY__ = "Utilities"


def setup(client, my_id):

    @client.on(
        events.NewMessage(from_users=my_id, pattern=r"^!block(?:\s+(\S+))?"))
    async def block_user(event):
        try:
            target = event.pattern_match.group(1)
            # Determine target user
            if event.is_reply:
                reply_msg = await event.get_reply_message()
                user = await reply_msg.get_sender()
            elif target:
                user = await client.get_entity(target)
            else:
                await event.edit(
                    "❌ Reply to a user or provide @username to block.")
                return

            await client(functions.contacts.BlockRequest(id=user.id))
            await event.edit(
                f"✅ Blocked user: {getattr(user, 'first_name', 'Unknown')} (@{getattr(user, 'username', 'N/A')})"
            )
        except Exception as e:
            await event.edit(f"❌ Error blocking user: {str(e)}")

    @client.on(
        events.NewMessage(from_users=my_id, pattern=r"^!unblock(?:\s+(\S+))?"))

# Com/calc.py
from telethon import events

__HELP__ = "!calc <num1> <op> <num2> → Calculate (supports + - × /). Must be used as reply."
__CATEGORY__ = "Utilities"

def setup(client, my_id):

    @client.on(events.NewMessage(from_users=my_id, pattern=r"^!calc\s+(.+)$"))
    async def calc_handler(event):
        try:
            if not event.is_reply:
                await event.edit(
                    "❌ Reply to a message and use !calc <num1> <op> <num2>")
                return

            expression = event.pattern_match.group(1).strip()
            # Normalize × to *
            expression = expression.replace("×", "*").replace("÷", "/")

            # Only allow digits, operators, dots, and spaces
            allowed = set("0123456789+-*/. ")
            if not all(c in allowed for c in expression):
                await event.edit("❌ Invalid expression")
                return

            # Evaluate safely
            try:
                result = eval(expression, {"__builtins__": {}})
            except Exception:
                await event.edit("❌ Calculation error")
                return

            # Delete command
            await event.delete()

            # Reply with result
            reply_msg = await event.get_reply_message()
            await client.send_message(event.chat_id,
                                      f"{result}",
                                      reply_to=reply_msg.id)

        except Exception as e:
            await event.edit(f"❌ Error: {str(e)}")

# Com/cauto.py
from telethon import events
from datetime import datetime
import re
import os
import json

__HELP__ = "!cauto ON/OFF → Auto solve math challenges in groups/DMs"

# Store state
AUTO_SOLVE = False
LOG_FILE = "mods/cauto_log.json"
os.makedirs("mods", exist_ok=True)

# Ensure log file exists
if not os.path.exists(LOG_FILE):
    with open(LOG_FILE, "w", encoding="utf-8") as f:
        json.dump([], f, indent=2)


def setup(client, my_id):
    global AUTO_SOLVE

    # Toggle ON/OFF
    @client.on(
        events.NewMessage(from_users=my_id, pattern=r"^!cauto\s+(on|off)$"))
    async def toggle_cauto(event):
        global AUTO_SOLVE
        state = event.pattern_match.group(1).lower()

        if state == "on":
            AUTO_SOLVE = True
            await event.edit("✅ Auto Challenge Solver: ON")
        else:
            AUTO_SOLVE = False
            await event.edit("❌ Auto Challenge Solver: OFF")

    # Logging helper
    async def log_challenge(chat_id, msg_id, challenge_type, equation, result):
        now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        log_entry = {
            "chat_id": chat_id,
            "message_id": msg_id,
            "type": challenge_type,
            "equation": equation,
            "result": result,
            "timestamp": now
        }
        try:
            with open(LOG_FILE, "r", encoding="utf-8") as f:
                data = json.load(f)
        except Exception:
            data = []

        data.append(log_entry)
        with open(LOG_FILE, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=2)

    # Auto-solver
    @client.on(events.NewMessage())
    async def auto_solver(event):
        global AUTO_SOLVE
        if not AUTO_SOLVE:
            return

        text = event.raw_text
        if not text:
            return

        # --------------------------
        # Old style: "چالش ریاضی"
        # --------------------------
        if "چالش ریاضی" in text and "=" in text:
            match = re.search(r"([\d\s\+\-\×\*/÷]+)= ?", text)
            if not match:
                return

            expression = match.group(1).strip()
            expression = expression.replace("×", "*").replace("÷", "/")

            try:
                result = eval(expression, {"__builtins__": {}})
                await client.send_message(event.chat_id,
                                          str(result),
                                          reply_to=event.id)
                await log_challenge(event.chat_id, event.id, "چالش ریاضی",
                                    expression, str(result))
            except Exception:
                pass

        # --------------------------
        # New style: "برای باز کردن جعبه حماسی"
        # --------------------------
        elif "برای باز کردن جعبه حماسی" in text:
            match = re.search(r"(\d+)\s*([\+\-\*/×÷])\s*(\d+)", text)
            if not match:
                return

            a, op, b = match.groups()
            op = op.replace("×", "*").replace("÷", "/")
            expr = f"{a}{op}{b}"

            try:
                result = str(eval(expr, {"__builtins__": {}}))
            except Exception:
                return

            # Click the correct button if exists
            clicked = False
            if event.message.buttons:
                for row in event.message.buttons:
                    for btn in row:
                        if btn.text.strip() == result:
                            try:
                                await event.click(btn)
                                clicked = True
                                break
                            except Exception:
                                pass
                    if clicked:
                        break

            await log_challenge(event.chat_id, event.id, "جعبه حماسی", expr,
                                result)

from telethon import events
import aiohttp
import os
import json
import uuid

__HELP__ = (
    "!checkexp <domain> → Scan domain for safe vulnerabilities (XSS, SQLi, headers, TLS, etc.)\n"
    "!checkexp del <ID> → Delete saved scan report")

SCAN_FOLDER = "mode/checkexp"
os.makedirs(SCAN_FOLDER, exist_ok=True)


def setup(client, my_id):

    async def safe_get(session, url):
        try:
            async with session.get(url, timeout=10,
                                   allow_redirects=True) as resp:
                text = await resp.text(errors='ignore')
                headers = dict(resp.headers)
                return resp.status, text, headers
        except:
            return None, None, {}

    @client.on(
        events.NewMessage(from_users=my_id,
                          pattern=r"^!checkexp\s+del\s+(\S+)$"))
    async def delete_scan(event):
        scan_id = event.pattern_match.group(1)
        file_path = os.path.join(SCAN_FOLDER, f"{scan_id}.txt")
        if os.path.exists(file_path):
            os.remove(file_path)
            await event.edit(f"✅ Scan report {scan_id} deleted.")
        else:
            await event.edit(f"❌ No scan report with ID {scan_id} found.")

    @client.on(
        events.NewMessage(from_users=my_id, pattern=r"^!checkexp\s+(\S+)$"))
    async def checkexp(event):
        domain = event.pattern_match.group(1).lower()
        scan_id = str(uuid.uuid4().int)[:12]  # unique short ID
        file_path = os.path.join(SCAN_FOLDER, f"{scan_id}.txt")

        await event.edit(
            f"🔍 Starting safe scan for {domain}...\nScan ID: {scan_id}")

        async with aiohttp.ClientSession() as session:
            results = []

            # Basic HTTP/S check
            for proto in ["https://", "http://"]:
                url = proto + domain
                status, text, headers = await safe_get(session, url)
                if status:
                    results.append(f"Protocol: {proto} Status: {status}")
                    results.append(f"Server headers: {headers}")
                    break

            # Header security checks
            security_headers = [
                "X-Frame-Options", "Content-Security-Policy",
                "Strict-Transport-Security", "X-Content-Type-Options"
            ]
            missing_headers = [h for h in security_headers if h not in headers]
            results.append(
                f"Missing security headers: {missing_headers or 'None'}")

            # Basic XSS safe test
            if text:
                xss_payload = "<script>alert(1)</script>"
                if "form" in text.lower() or "input" in text.lower():
                    results.append(
                        "⚠️ Forms detected, potential XSS targets (safe payload not executed)"
                    )

            # SQLi safe check
            sqli_test = "' OR '1'='1"
            if "sql" in text.lower() or "database" in text.lower():
                results.append(
                    "⚠️ Possible SQLi points detected (safe payload not executed)"
                )

            # Robots.txt
            robots_url = f"https://{domain}/robots.txt"
            status, robots_text, _ = await safe_get(session, robots_url)
            if status == 200 and robots_text:
                results.append(f"robots.txt found:\n{robots_text[:300]}..."
                               )  # limit preview

            # Sitemap.xml
            sitemap_url = f"https://{domain}/sitemap.xml"
            status, sitemap_text, _ = await safe_get(session, sitemap_url)
            if status == 200 and sitemap_text:
                results.append(f"sitemap.xml found:\n{sitemap_text[:300]}..."
                               )  # limit preview

            # TLS / HTTPS check
            if proto == "https://":

from telethon import events
import asyncio
import os
import json

__HELP__ = (
    "!countdown <number><s/m/h> <reason> → Start a countdown in this chat\n"
    "!countdown stop → Stop the countdown running in this chat")
__CATEGORY__ = "Utilities"

# -----------------------------
# Ensure mods folder exists
# -----------------------------
os.makedirs("mods", exist_ok=True)
CD_FILE = "mods/cd.json"

# -----------------------------
# Active countdowns dict
# -----------------------------
active_countdowns = {
}  # chat_id: {"remaining": seconds, "reason": str, "msg": Message, "task": Task}


# -----------------------------
# Setup function for Telethon
# -----------------------------
def setup(client, my_id):

    def format_hms(seconds: int) -> str:
        h = seconds // 3600
        m = (seconds % 3600) // 60
        s = seconds % 60
        return f"{h:02d}:{m:02d}:{s:02d}"

    async def save_cd():
        data = {}
        for chat_id, info in active_countdowns.items():
            data[str(chat_id)] = {
                "remaining": info["remaining"],
                "reason": info["reason"]
            }
        with open(CD_FILE, "w") as f:
            json.dump(data, f)

    async def load_cd():
        if not os.path.exists(CD_FILE):
            return {}
        try:
            with open(CD_FILE, "r") as f:
                return json.load(f)
        except:
            return {}

    async def run_countdown(chat_id: int, total_seconds: int, reason: str,
                            msg):
        active_countdowns[chat_id] = {
            "remaining": total_seconds,
            "reason": reason,
            "msg": msg,
            "task": asyncio.current_task()
        }
        remaining = total_seconds
        try:
            while remaining > 0:
                display = format_hms(remaining)
                try:
                    await msg.edit(f"⏳ Countdown: {display}")
                except:
                    pass
                active_countdowns[chat_id]["remaining"] = remaining
                await save_cd()
                await asyncio.sleep(1)
                remaining -= 1
            try:
                await msg.edit(f"✅ Countdown finished! Reason: {reason}")
            except:
                pass
        except asyncio.CancelledError:
            try:
                await msg.edit("❌ Countdown stopped!")
            except:
                pass
        finally:
            active_countdowns.pop(chat_id, None)
            await save_cd()

    # -----------------------------
    # Resume countdowns from cd.json
    # -----------------------------
    async def resume_cds():
        data = await load_cd()
        for chat_id_str, info in data.items():
            chat_id = int(chat_id_str)
            remaining = info.get("remaining", 0)
            reason = info.get("reason", "")
            msg = await client.send_message(
                chat_id, f"⏳ Countdown resumed: {format_hms(remaining)}")
            asyncio.create_task(run_countdown(chat_id, remaining, reason, msg))

    asyncio.create_task(resume_cds())

    # -----------------------------
    # Start a countdown
    # -----------------------------
    @client.on(
        events.NewMessage(from_users=my_id,
                          pattern=r"^!countdown\s+(\d+)([smh])\s+(.+)$"))
    async def countdown(event):
        chat_id = event.chat_id
        if chat_id in active_countdowns:
            await event.reply(
                "❌ A countdown is already running in this chat! Stop it first with !countdown stop."
            )
            return

        number = int(event.pattern_match.group(1))
        unit = event.pattern_match.group(2).lower()
        reason = event.pattern_match.group(3)

        if unit == "s":
            total_seconds = number
        elif unit == "m":
            total_seconds = number * 60
        elif unit == "h":
            total_seconds = number * 3600
        else:
            await event.reply("❌ Invalid unit! Use s, m, or h.")
            return

        msg = await event.reply(f"⏳ Countdown: {format_hms(total_seconds)}")
        asyncio.create_task(run_countdown(chat_id, total_seconds, reason, msg))

        await event.delete()  # remove the command

    # -----------------------------
    # Stop countdown
    # -----------------------------
    @client.on(
        events.NewMessage(from_users=my_id, pattern=r"^!countdown stop$"))
    async def stop(event):
        chat_id = event.chat_id
        info = active_countdowns.get(chat_id)
        if info:
            task = info.get("task")
            msg = info.get("msg")
            if msg:
                await msg.edit("❌ Countdown stopping...")
            if task:
                task.cancel()
                try:
                    await task
                except asyncio.CancelledError:
                    pass
            active_countdowns.pop(chat_id, None)
            await save_cd()
            await event.delete()
        else:
            await event.reply(
                "❌ No countdown is currently running in this chat!")

# Com/decodeqr.py
import aiohttp
from telethon import events
from io import BytesIO

__HELP__ = "!decodeqr → Decode a QR code from a replied image using a free API"
__CATEGORY__ = "Utilities"

def setup(client, my_id):

    @client.on(events.NewMessage(from_users=my_id, pattern=r"^!decodeqr"))
    async def decode_qr(event):
        try:
            if not event.is_reply:
                await event.reply("❌ Reply to an image containing a QR code.")
                return

            reply_msg = await event.get_reply_message()
            if not reply_msg.media:
                await event.reply("❌ The replied message has no media.")
                return

            bio = BytesIO()
            await reply_msg.download_media(file=bio)
            bio.seek(0)

            async with aiohttp.ClientSession() as session:
                data = aiohttp.FormData()
                data.add_field("file",
                               bio,
                               filename="qr.png",
                               content_type="image/png")

                async with session.post(
                        "https://api.qrserver.com/v1/read-qr-code/",
                        data=data) as resp:
                    if resp.status != 200:
                        await event.reply(
                            f"❌ API request failed with status {resp.status}")
                        return
                    result = await resp.json()

            # Parse the response
            decoded_texts = []
            for item in result:
                for symbol in item.get("symbol", []):
                    text = symbol.get("data")
                    if text:
                        decoded_texts.append(text)

            if not decoded_texts:
                await event.reply("❌ No QR code found in the image.")
                return

            await event.reply("🔓 Decoded QR Code:\n" +
                              "\n".join(decoded_texts))

        except Exception as e:
            await event.reply(f"❌ Error: {str(e)}")

# Com/dnsprobe.py
from telethon import events
import os
import json
import asyncio
import dns.resolver
from datetime import datetime

__HELP__ = "!dnsprobe <domain1> <domain2> ... → Bulk DNS record enumerator (A, AAAA, MX, TXT, CNAME)\n!dnsprobe del <ID> → Delete a previous probe file by ID"

MODE_FOLDER = "mode/dns"
os.makedirs(MODE_FOLDER, exist_ok=True)
ID_FILE = os.path.join(MODE_FOLDER, "dns_id.txt")
if not os.path.exists(ID_FILE):
    with open(ID_FILE, "w") as f:
        f.write("0")

active_dns_tasks = {}  # id: task_info


def setup(client, my_id):

    def get_next_id():
        with open(ID_FILE, "r") as f:
            last_id = int(f.read().strip())
        next_id = last_id + 1
        with open(ID_FILE, "w") as f:
            f.write(str(next_id))
        return next_id

    async def probe_domain(domain):
        records = {}
        for rtype in ["A", "AAAA", "MX", "TXT", "CNAME"]:
            try:
                answers = dns.resolver.resolve(domain, rtype, lifetime=5)
                records[rtype] = [str(r.to_text()) for r in answers]
            except Exception:
                records[rtype] = []
        return records

    @client.on(
        events.NewMessage(from_users=my_id, pattern=r"^!dnsprobe(?:\s+(.+))?"))
    async def dnsprobe_handler(event):
        try:
            args = event.pattern_match.group(1)
            if not args:
                await event.reply(
                    "❌ Provide at least one domain or use !dnsprobe del <ID>")
                return

            parts = args.split()
            if parts[0].lower() == "del":
                del_id = int(parts[1])
                # Delete previous file
                for filename in os.listdir(MODE_FOLDER):
                    if filename.startswith(f"{del_id}_dnsprobe_"):
                        os.remove(os.path.join(MODE_FOLDER, filename))
                        await event.edit(
                            f"✅ Deleted DNS probe file with ID {del_id}")
                        return
                await event.edit(f"❌ No DNS probe found with ID {del_id}")
                return

            domains = parts
            probe_id = get_next_id()
            now = datetime.now().strftime("%Y-%m-%d_%H-%M-%S")
            temp_file = os.path.join(MODE_FOLDER,
                                     f"{probe_id}_dnsprobe_{now}.txt")

            await event.edit(
                f"⏳ Probing {len(domains)} domain(s)... Task ID: {probe_id}")

            results = []
            for domain in domains:
                records = await probe_domain(domain)
                result_str = f"Domain: {domain}\n"
                for rtype, values in records.items():
                    result_str += f"  {rtype}: {', '.join(values) if values else 'None'}\n"
                result_str += "\n"
                results.append(result_str)

            with open(temp_file, "w", encoding="utf-8") as f:
                f.writelines(results)

            active_dns_tasks[probe_id] = {
                "file": temp_file,
                "domains": domains,
                "timestamp": now
            }

            await client.send_file(
                event.chat_id,
                temp_file,
                caption=f"✅ DNS Probe complete. Task ID: {probe_id}",
                reply_to=event.id)
            await event.delete()

        except Exception as e:
            await event.reply(f"❌ Error: {str(e)}")

from telethon import events
import hashlib
import os
import datetime

__HELP__ = "!fileinfo (reply) → Get detailed info about a replied file (size, type, hashes, etc.)"
__CATEGORY__ = "Utilities"


def setup(client, my_id):

    @client.on(events.NewMessage(from_users=my_id, pattern=r"^!fileinfo$"))
    async def fileinfo(event):
        if not event.is_reply:
            await event.reply("❌ Reply to a file to get its info.")
            return

        reply_msg = await event.get_reply_message()
        if not reply_msg.media:
            await event.reply("❌ The replied message doesn’t contain a file.")
            return

        try:
            file = reply_msg.file
            if not file:
                await event.reply("❌ No file information found.")
                return

            # Download file temporarily to calculate hashes
            path = await reply_msg.download_media(file="./temp_fileinfo")
            md5_hash = sha1_hash = sha256_hash = "N/A"
            if path and os.path.exists(path):
                with open(path, "rb") as f:
                    data = f.read()
                    md5_hash = hashlib.md5(data).hexdigest()
                    sha1_hash = hashlib.sha1(data).hexdigest()
                    sha256_hash = hashlib.sha256(data).hexdigest()
                os.remove(path)

            # Human-readable size
            def human_size(size):
                if size is None:
                    return "Unknown"
                for unit in ["B", "KB", "MB", "GB", "TB"]:
                    if size < 1024:
                        return f"{size:.2f} {unit}"
                    size /= 1024
                return f"{size:.2f} PB"

            # Format message
            msg = "📂 **File Information**\n\n"
            if file.name:
                msg += f"**Name:** `{file.name}`\n"
            if file.size:
                msg += f"**Size:** {human_size(file.size)} ({file.size} bytes)\n"
            if file.mime_type:
                msg += f"**MIME Type:** `{file.mime_type}`\n"
            if file.id:
                msg += f"**File ID:** `{file.id}`\n"
            if reply_msg.date:
                msg += f"**Upload Date:** `{reply_msg.date.strftime('%Y-%m-%d %H:%M:%S')}`\n"

            # Try to extract document-specific fields
            if hasattr(reply_msg.media, "document"):
                doc = reply_msg.media.document
                if hasattr(doc, "dc_id"):
                    msg += f"**Data Center ID:** `{doc.dc_id}`\n"
                if hasattr(doc, "mime_type"):
                    msg += f"**MIME Type (Doc):** `{doc.mime_type}`\n"

            # Media specific info
            if file.width and file.height:
                msg += f"**Dimensions:** {file.width}×{file.height}\n"
            if file.duration:
                duration = str(datetime.timedelta(seconds=file.duration))
                msg += f"**Duration:** {duration}\n"

            # Hashes
            msg += "\n🔑 **Hashes:**\n"
            msg += f"MD5: `{md5_hash}`\n"
            msg += f"SHA1: `{sha1_hash}`\n"
            msg += f"SHA256: `{sha256_hash}`\n"

            await event.reply(msg)

        except Exception as e:
            await event.reply(f"❌ Error: {str(e)}")

from telethon import events
import os
from datetime import datetime
import pytz

__HELP__ = "!finduser <username/ID> → Gather all messages sent by a user in all group logs."
__CATEGORY__ = "Information"
GROUP_FOLDER = "Logs/Groups"
tz = pytz.timezone("Asia/Tehran")


def setup(client, my_id):

    @client.on(
        events.NewMessage(from_users=my_id,
                          pattern=r"^!finduser(?:\s+(here|.+))?$"))
    async def finduser(event):
        try:
            args = event.pattern_match.group(1)
            send_here = False
            target_user_id = None

            # -------------------------------
            # Reply case
            # -------------------------------
            if event.is_reply:
                reply_msg = await event.get_reply_message()
                sender = await reply_msg.get_sender()
                target_user_id = sender.id

                if args and args.lower() == "here":
                    send_here = True  # send in command chat
                else:
                    send_here = False  # send in Saved Messages

            # -------------------------------
            # Mention/ID case
            # -------------------------------
            elif args:
                parts = args.split()
                if parts[0].lower() == "here":
                    send_here = True
                    if len(parts) > 1:
                        user_str = parts[1]
                        if user_str.isdigit():
                            target_user_id = int(user_str)
                        else:
                            try:
                                user_entity = await client.get_entity(user_str)
                                target_user_id = user_entity.id
                            except:
                                await event.reply(
                                    f"❌ Cannot find user '{user_str}'")
                                return
                else:
                    send_here = False
                    user_str = parts[0]
                    if user_str.isdigit():
                        target_user_id = int(user_str)
                    else:
                        try:
                            user_entity = await client.get_entity(user_str)
                            target_user_id = user_entity.id
                        except:
                            await event.reply(
                                f"❌ Cannot find user '{user_str}'")
                            return
            else:
                await event.reply(
                    "❌ Please specify a user or reply to a message.")
                return

            if target_user_id is None:
                await event.reply("❌ Could not determine user ID.")
                return

            log_lines = []
            # -------------------------------
            # Iterate all group files
            # -------------------------------
            for filename in os.listdir(GROUP_FOLDER):
                if not filename.endswith(".txt"):
                    continue
                file_path = os.path.join(GROUP_FOLDER, filename)
                with open(file_path, "r", encoding="utf-8") as f:
                    lines = f.readlines()

                group_name = lines[0].strip() if lines else "Unknown Group"
                group_msgs = []
                skip_next = False
                for i, line in enumerate(lines[1:], start=1):
                    if skip_next:
                        skip_next = False
                        continue
                    if f"({target_user_id} -" in line:
                        collected = [line.strip()]
                        if i + 1 < len(lines):
                            next_line = lines[i + 1].strip()
                            if next_line.startswith(
                                    "Edited") or next_line.startswith(
                                        "Deleted"):
                                collected.append(next_line)
                                skip_next = True
                        group_msgs.extend(collected)

                if group_msgs:
                    log_lines.append(f"{group_name}:\n" +
                                     "\n".join(group_msgs) + "\n.\n")

            if not log_lines:
                await event.edit(
                    f"❌ No messages found for **{target_user_id}**")
                return

            # -------------------------------
            # Create temp file
            # -------------------------------
            now = datetime.now(tz).strftime("%Y-%m-%d_%H-%M-%S")
            temp_file = f"{target_user_id}_{now}.txt"
            with open(temp_file, "w", encoding="utf-8") as f:
                f.writelines("\n".join(log_lines))

            # -------------------------------
            # Send file to proper chat
            # -------------------------------
            target_chat = event.chat_id if send_here else "me"
            await event.client.send_file(
                target_chat,
                temp_file,
                caption=f"Sent the Log file of **{target_user_id}**",
                reply_to=event.id if send_here else None)
            await event.edit(f"Sent the Log file of **{target_user_id}**")
            os.remove(temp_file)

        except Exception as e:
            await event.edit(f"❌ Error: {str(e)}")

                        p.strip() for p in help_text.split("\n") if p.strip()
                    ]

                    folder = "Utilities"
                    if any(
                            k in modulename for k in
                        ["block", "finduser", "ginfo", "whois", "info", "log"
                         ]):
                        folder = "Information"
                    elif any(k in modulename for k in
                             ["help", "ping", "setreaction", "update"]):
                        folder = "Setup"

                            split = part.split(" ", 1)
                            cmd = f"**`{split[0]}`**"

from telethon import events
from PIL import Image, ImageDraw, ImageFont
from io import BytesIO
import os

__HELP__ = "!imgwatermark <text> or reply → Add watermark text to an image"

os.makedirs("temp", exist_ok=True)


def setup(client, my_id):

    @client.on(
        events.NewMessage(from_users=my_id,
                          pattern=r"^!imgwatermark(?:\s+(.+))?"))
    async def imgwatermark(event):
        try:
            # Get watermark text
            text = event.pattern_match.group(1)
            if not text and event.is_reply:
                reply_msg = await event.get_reply_message()
                text = reply_msg.text or reply_msg.message
            if not text:
                await event.reply(
                    "❌ Provide watermark text or reply to a message with text."
                )
                return

            # Check if the command is a reply to an image
            if not event.is_reply:
                await event.reply("❌ Reply to an image to add a watermark.")
                return

            reply_msg = await event.get_reply_message()
            if not reply_msg.media:
                await event.reply("❌ Replied message has no media.")
                return

            # Download image to memory
            bio = BytesIO()
            await reply_msg.download_media(file=bio)
            bio.seek(0)

            # Open image
            img = Image.open(bio).convert("RGBA")

            # Create watermark layer
            txt_layer = Image.new("RGBA", img.size, (255, 255, 255, 0))
            draw = ImageDraw.Draw(txt_layer)
            font_size = max(20, img.width // 15)
            font = ImageFont.truetype(
                "arial.ttf", font_size) if os.path.exists(
                    "arial.ttf") else ImageFont.load_default()

            # Calculate text size using textbbox()
            bbox = draw.textbbox((0, 0), text, font=font)
            text_width = bbox[2] - bbox[0]
            text_height = bbox[3] - bbox[1]

            # Position: bottom-right corner
            x = img.width - text_width - 10
            y = img.height - text_height - 10

            # Draw text
            draw.text((x, y), text, font=font,
                      fill=(255, 255, 255, 128))  # semi-transparent white

            # Combine original image with watermark
            watermarked = Image.alpha_composite(img, txt_layer)

            # Save to BytesIO
            out_bio = BytesIO()
            watermarked.convert("RGB").save(out_bio, "JPEG")
            out_bio.seek(0)

            # Send watermarked image
            await client.send_file(
                event.chat_id,
                out_bio,
                caption=f"💧 Watermarked Image with text: {text}",
                reply_to=event.id)

        except Exception as e:
            await event.reply(f"❌ Error: {str(e)}")

__CATEGORY__ = "Information"

def human_readable_size(size_bytes):
    """Convert bytes to a readable format"""
    for unit in ['B', 'KB', 'MB', 'GB', 'TB']:
        if size_bytes < 1024:
            return f"{size_bytes:.2f} {unit}"
        size_bytes /= 1024
    return f"{size_bytes:.2f} PB"


def setup(client, my_id):

    @client.on(events.NewMessage(from_users=my_id, pattern=r"^!logdata$"))
    async def logdata(event):
        try:
            # Edit the command to show "Calculating..."
            await event.edit("📊 Calculating log data...")

            total_files = 0
            total_size = 0

            for folder in SUBFOLDERS:
                path = os.path.join(LOG_FOLDER, folder)
                if os.path.exists(path):
                    for root, dirs, files in os.walk(path):
                        total_files += len(files)
                        for f in files:
                            total_size += os.path.getsize(os.path.join(
                                root, f))

            readable_size = human_readable_size(total_size)

            # Edit the command message with the final result
            await event.edit(
                f"📁 There are {total_files} log files saved in '{LOG_FOLDER}'.\n💾 Storage used: {readable_size}"
            )

        except Exception as e:
            await event.edit(f"❌ Failed to get log data: {str(e)}")

def setup(client, my_id):

        events.NewMessage(
            from_users=my_id,
            pattern=r"^!logfile\s+(Channel|Group)\s+(-?\d+)$"))
    async def send_log_file(event):
        folder_map = {
            "PV": PV_FOLDER,
            "Group": GROUP_FOLDER,
            "Channel": CHANNEL_FOLDER
        }

        parts = event.text.split()
        log_type = parts[1]
        number_id = parts[2]

        folder = folder_map.get(log_type)

        filename = f"{number_id}.txt"
        file_path = None
        for f in os.listdir(folder):
            if f.strip() == filename:
                file_path = os.path.join(folder, f)
                break

        if not file_path or not os.path.isfile(file_path):
            await event.edit(
                f"❌ Log file not found: {filename}\nCheck folder and ID.")
            return

        # Send the file
        await client.send_file(int(my_id),
                               file_path,
                               caption=f"Log file: {filename}")

            return

# Com/portsweep.py
from telethon import events
import asyncio
import socket
import os
import json
from datetime import datetime

__HELP__ = "!portsweep <host> [start_port-end_port] → Scan ports safely and get open/filtered list.\n!portsweep del <ID> → Delete saved scan result."

SAVE_FOLDER = "mode/portsweep"
os.makedirs(SAVE_FOLDER, exist_ok=True)

active_scans = {
}  # id: {"host": host, "ports": list, "task": Task, "file": filename}


def setup(client, my_id):

    async def scan_ports(host, ports, scan_id):
        open_ports = []
        for port in ports:
            try:
                conn = asyncio.open_connection(host, port)
                reader, writer = await asyncio.wait_for(conn, timeout=1)
                open_ports.append(port)
                writer.close()
                await writer.wait_closed()
            except:
                continue

        now = datetime.now().strftime("%Y-%m-%d_%H-%M-%S")
        filename = f"{SAVE_FOLDER}/portsweep_{scan_id}.txt"
        with open(filename, "w", encoding="utf-8") as f:
            f.write(
                f"Host: {host}\nScan ID: {scan_id}\nDate: {now}\nOpen Ports:\n"
            )
            for p in open_ports:
                f.write(f"{p}\n")

        await client.send_file(
            client.me if scan_id.startswith("dm") else scan_id,
            filename,
            caption=f"✅ Portsweep completed for {host}. Scan ID: {scan_id}")
        active_scans.pop(scan_id, None)

    @client.on(
        events.NewMessage(from_users=my_id, pattern=r"^!portsweep\s+(.+)"))
    async def portsweep_cmd(event):
        args = event.pattern_match.group(1).strip()
        try:
            # Delete command
            await event.delete()

            # Delete existing scan
            if args.startswith("del"):
                del_id = args.split()[1]
                info = active_scans.get(del_id)
                if info:
                    file = info.get("file")
                    task = info.get("task")
                    if task:
                        task.cancel()
                    if file and os.path.exists(file):
                        os.remove(file)
                    active_scans.pop(del_id)
                    await event.reply(f"✅ Deleted portsweep result {del_id}")
                    return
                else:
                    await event.reply(f"❌ Scan ID {del_id} not found")
                    return

            # Parse host and ports
            parts = args.split()

    def log_url(original, short, service):
        data = []
        if os.path.exists(LOG_FILE):
            with open(LOG_FILE, "r", encoding="utf-8") as f:
                try:
                    data = json.load(f)
                except:
                    data = []
        data.append({"original": original, "short": short, "service": service})
        with open(LOG_FILE, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=2)

    async def tinyurl_shorten(url):
        async with aiohttp.ClientSession() as session:
            async with session.get(
                    f"http://tinyurl.com/api-create.php?url={url}") as resp:
                if resp.status != 200:
                    return None
                return await resp.text()

    async def bitly_shorten(url):
        if not BITLY_TOKEN:
            return None
        headers = {
            "Authorization": f"Bearer {BITLY_TOKEN}",
            "Content-Type": "application/json"
        }
        async with aiohttp.ClientSession() as session:
            async with session.post("https://api-ssl.bitly.com/v4/shorten",
                                    headers=headers,
                                    json={"long_url": url}) as resp:
                if resp.status != 200:
                    return None
                data = await resp.json()
                return data.get("link")

    @client.on(
        events.NewMessage(from_users=my_id,
                          pattern=r"^!shorturl(?:\s+(\w+))?(?:\s+(.+))?"))
    async def shorturl(event):
        try:
            service = "tinyurl"
            url = event.pattern_match.group(2)
            arg1 = event.pattern_match.group(1)

            # Check if first argument is a service name
            if arg1 and arg1.lower() in ["tinyurl", "bitly"]:
                service = arg1.lower()
            elif arg1 and not url:
                url = arg1

            # If no URL in command, check reply
            if not url and event.is_reply:
                reply_msg = await event.get_reply_message()
                url = reply_msg.text or reply_msg.message or ""

            if not url:
                await event.reply(
                    "❌ Provide a URL or reply to a message containing one.")
                return

            if not url.startswith("http"):
                url = "http://" + url

            # Shorten
            if service == "bitly":
                short_link = await bitly_shorten(url)
                if not short_link:
                    await event.reply(
                        "❌ Failed to shorten using Bitly. Check token or URL.")
                    return
            else:
                short_link = await tinyurl_shorten(url)
                if not short_link:
                    await event.reply("❌ Failed to shorten using TinyURL.")
                    return

            # Reply with shortened URL
            await event.reply(
                f"🔗 Shortened URL ({service.title()}):\n{short_link}")

            # Log it
            log_url(url, short_link, service)

        except Exception as e:
            await event.reply(f"❌ Error: {str(e)}")

from telethon import events
import aiohttp
import asyncio
import os
import json
from datetime import datetime
import socket

__HELP__ = (
    "!subenum <domain> → Enumerate active subdomains and send as TXT file\n"
    "!subenum del <ID> → Delete a previously saved subdomain TXT file")
__CATEGORY__ = "Recon & Discovery"

LOG_DIR = "mode/subenum_logs"
ACTIVE_FILE = "mode/subenum_active.json"
os.makedirs(LOG_DIR, exist_ok=True)
os.makedirs("mode", exist_ok=True)


def setup(client, my_id):

    async def fetch_ct_subdomains(domain):
        url = f"https://crt.sh/?q=%25.{domain}&output=json"
        try:
            async with aiohttp.ClientSession() as session:
                async with session.get(url, timeout=20) as resp:
                    if resp.status != 200:
                        return []
                    data = await resp.json()
                    subdomains = set()
                    for entry in data:
                        name = entry.get("name_value", "")
                        for sub in name.split("\n"):
                            if sub.endswith(domain):
                                subdomains.add(sub.strip())
                    return list(subdomains)
        except:
            return []

    def check_dns(subdomain):
        try:
            ip = socket.gethostbyname(subdomain)
            return ip
        except:
            return None

    def load_active():
        if os.path.exists(ACTIVE_FILE):
            with open(ACTIVE_FILE, "r", encoding="utf-8") as f:
                try:
                    return json.load(f)
                except:
                    return {}
        return {}

    def save_active(active_data):
        with open(ACTIVE_FILE, "w", encoding="utf-8") as f:
            json.dump(active_data, f, indent=2)

    @client.on(
        events.NewMessage(from_users=my_id, pattern=r"^!subenum(?:\s+(.+))?"))
    async def subenum_handler(event):
        try:
            arg = event.pattern_match.group(1)
            if not arg and event.is_reply:
                reply_msg = await event.get_reply_message()
                arg = (reply_msg.text or "").strip()

            if not arg:
                await event.reply(
                    "❌ Provide a domain or reply to a message containing one.")
                return

            # Delete command
            await event.delete()

            # Handle delete command
            if arg.startswith("del"):
                parts = arg.split()
                if len(parts) != 2 or not parts[1].isdigit():
                    await event.reply("❌ Usage: !subenum del <ID>")
                    return
                del_id = parts[1]
                active_data = load_active()
                entry = active_data.get(del_id)
                if entry and os.path.exists(entry["file"]):
                    os.remove(entry["file"])
                    del active_data[del_id]
                    save_active(active_data)
                    await event.reply(
                        f"✅ Deleted subdomain file with ID {del_id}")
                else:
                    await event.reply(f"❌ File with ID {del_id} not found")
                return

            domain = arg
            active_data = load_active()

            # Check if already scanned
            for task_id, info in active_data.items():
                if info.get("domain") == domain and os.path.exists(
                        info.get("file", "")):
                    await event.reply(
                        f"📄 Already scanned **{domain}**, sending saved file..."
                    )
                    await client.send_file(
                        event.chat_id,
                        info["file"],
                        caption=f"Subdomains for **{domain}** (ID: {task_id})")
                    return

            await event.reply(f"🔍 Scanning subdomains for **{domain}**...")

            subdomains = await fetch_ct_subdomains(domain)
            active_subs = []
            for sub in subdomains:
                ip = await asyncio.get_event_loop().run_in_executor(
                    None, check_dns, sub)
                if ip:
                    active_subs.append(f"{sub} → {ip}")

            if not active_subs:
                await event.reply(
                    f"❌ No active subdomains found for **{domain}**")
                return

            # Save to TXT file
            now = datetime.now().strftime("%Y%m%d%H%M%S")
            task_id = int(now)
            file_path = os.path.join(LOG_DIR, f"{domain}_{task_id}.txt")
            with open(file_path, "w", encoding="utf-8") as f:
                f.write("\n".join(active_subs))

            # Save active data
            active_data[task_id] = {
                "domain": domain,
                "file": file_path,
                "timestamp": now
            }
            save_active(active_data)

            await client.send_file(
                event.chat_id,
                file_path,
                caption=f"✅ Active subdomains for **{domain}** (ID: {task_id})"
            )

        except Exception as e:
            await event.reply(f"❌ Error: {str(e)}")

    async def collect_sysinfo():
        try:
            now = datetime.now(TZ)
            # CPU
            cpu_freq = psutil.cpu_freq()
            cpu_percent_per_core = psutil.cpu_percent(percpu=True)
            cpu_count = psutil.cpu_count(logical=True)
            cpu_cores = psutil.cpu_count(logical=False)

            # Memory
            mem = psutil.virtual_memory()
            swap = psutil.swap_memory()

            # Disks
            disks = []
            for part in psutil.disk_partitions(all=False):
                try:
                    usage = psutil.disk_usage(part.mountpoint)
                    disks.append({
                        "device": part.device,
                        "mount": part.mountpoint,
                        "used": usage.used,
                        "total": usage.total,
                        "percent": usage.percent
                    })
                except:
                    continue

            # Network
            net_io = psutil.net_io_counters()
            conns = len(psutil.net_connections())

            # Uptime
            boot_time = datetime.fromtimestamp(psutil.boot_time(), tz=TZ)
            uptime_sec = (now - boot_time).total_seconds()
            days, rem = divmod(uptime_sec, 86400)
            hours, rem = divmod(rem, 3600)
            minutes, _ = divmod(rem, 60)

            # GPU
            gpus = []
            if GPUtil:
                try:
                    for gpu in GPUtil.getGPUs():
                        gpus.append({
                            "name": gpu.name,
                            "load": gpu.load * 100,
                            "mem_used": gpu.memoryUsed,
                            "mem_total": gpu.memoryTotal,
                            "temp": gpu.temperature
                        })
                except:
                    pass

            sysinfo = {
                "timestamp": now.strftime("%Y-%m-%d %H:%M:%S"),
                "cpu": {
                    "cores": cpu_cores,
                    "threads": cpu_count,
                    "freq": cpu_freq.current if cpu_freq else None,
                    "per_core_usage": cpu_percent_per_core
                },
                "memory": {
                    "total": mem.total,
                    "used": mem.used,
                    "percent": mem.percent,
                    "swap_total": swap.total,
                    "swap_used": swap.used,
                    "swap_percent": swap.percent
                },
                "disks": disks,
                "network": {
                    "sent": net_io.bytes_sent,
                    "recv": net_io.bytes_recv,
                    "connections": conns
                },
                "uptime":
                f"{int(days)}d {int(hours)}h {int(minutes)}m (since {boot_time.strftime('%Y-%m-%d %H:%M:%S')} IRST)",
                "platform":
                f"{platform.system()} {platform.release()} ({platform.machine()})",
                "python": platform.python_version(),
                "gpus": gpus
            }

            # Save log

# Com/sysmon.py
import time
import psutil
import platform
import os
import asyncio
import json
from telethon import events

__HELP__ = "!sysmon start/stop/export → Live system monitor that updates every 5s, logs stats, and can export logs."
__CATEGORY__ = "Utilities"

START_TIME = time.time()
active_monitors = {}  # chat_id -> task

LOG_DIR = "mode/sys"
LOG_FILE = os.path.join(LOG_DIR, "sysmon.json")
os.makedirs(LOG_DIR, exist_ok=True)


def format_stats():
    uptime_seconds = int(time.time() - START_TIME)
    days, remainder = divmod(uptime_seconds, 86400)
    hours, remainder = divmod(remainder, 3600)
    minutes, seconds = divmod(remainder, 60)
    uptime_str = f"{days}d {hours}h {minutes}m {seconds}s"

    cpu_usage = psutil.cpu_percent(interval=0.5)
    cpu_count = psutil.cpu_count(logical=True)

    memory = psutil.virtual_memory()
    mem_usage = (f"{memory.used // (1024**2)}MB / "
                 f"{memory.total // (1024**2)}MB ({memory.percent}%)")

    swap = psutil.swap_memory()
    swap_usage = (f"{swap.used // (1024**2)}MB / "
                  f"{swap.total // (1024**2)}MB ({swap.percent}%)"
                  if swap.total > 0 else "N/A")

    load_avg = "N/A"
    if hasattr(os, "getloadavg"):
        la1, la5, la15 = os.getloadavg()
        load_avg = f"{la1:.2f}, {la5:.2f}, {la15:.2f}"

    disk = psutil.disk_usage("/")
    disk_usage = (f"{disk.used // (1024**3)}GB / "
                  f"{disk.total // (1024**3)}GB ({disk.percent}%)")

    processes = len(psutil.pids())
    sys_info = f"{platform.system()} {platform.release()} ({platform.machine()})"

    stats = {
        "uptime": uptime_str,
        "cpu_usage": f"{cpu_usage}%",
        "cpu_cores": cpu_count,
        "memory": mem_usage,
        "swap": swap_usage,
        "load_avg": load_avg,
        "disk": disk_usage,
        "processes": processes,
        "system": sys_info,
        "timestamp": time.strftime("%Y-%m-%d %H:%M:%S", time.localtime())
    }

    return stats


def save_stats(stats):
    """Save stats to JSON, ensure it's a list."""
    data = []
    if os.path.exists(LOG_FILE):
        try:
            with open(LOG_FILE, "r", encoding="utf-8") as f:
                loaded = json.load(f)
                if isinstance(loaded, list):
                    data = loaded
                else:
                    # If file contains dict or something else, reset to list
                    data = []
        except Exception:
            data = []

    data.append(stats)

    with open(LOG_FILE, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=2)


def pretty_stats(stats):
    return (
        f"⏱ **Uptime:** `{stats['uptime']}`\n\n"
        f"💻 **System:** `{stats['system']}`\n"
        f"🧮 **CPU:** `{stats['cpu_usage']}` of {stats['cpu_cores']} cores\n"
        f"📦 **Memory:** `{stats['memory']}`\n"
        f"💾 **Swap:** `{stats['swap']}`\n"
        f"📊 **Load Avg (1/5/15m):** `{stats['load_avg']}`\n"
        f"💽 **Disk:** `{stats['disk']}`\n"
        f"⚙️ **Processes:** `{stats['processes']}`\n"
        f"🕒 **Logged at:** `{stats['timestamp']}`")


def setup(client, my_id):

    async def monitor_task(chat_id, msg):
        try:
            while chat_id in active_monitors:
                stats = format_stats()
                save_stats(stats)  # log into file
                await msg.edit(pretty_stats(stats))
                await asyncio.sleep(5)
        except asyncio.CancelledError:
            pass
        except Exception as e:
            await msg.edit(f"❌ Monitor error: {str(e)}")
        finally:
            active_monitors.pop(chat_id, None)

    @client.on(events.NewMessage(from_users=my_id, pattern=r"^!sysmon start$"))
    async def start_sysmon(event):
        chat_id = event.chat_id
        if chat_id in active_monitors:
            await event.reply("⚠️ SysMon is already running in this chat.")
            return

        msg = await event.reply("🖥 Starting system monitor...")
        task = asyncio.create_task(monitor_task(chat_id, msg))
        active_monitors[chat_id] = task
        await event.delete()

    @client.on(events.NewMessage(from_users=my_id, pattern=r"^!sysmon stop$"))
    async def stop_sysmon(event):
        chat_id = event.chat_id
        if chat_id not in active_monitors:
            await event.reply("⚠️ SysMon is not running here.")
            return

        task = active_monitors.pop(chat_id)
        task.cancel()
        await event.reply("🛑 SysMon stopped.")
        await event.delete()

    @client.on(events.NewMessage(from_users=my_id,
                                 pattern=r"^!sysmon export$"))
    async def export_sysmon(event):
        if not os.path.exists(LOG_FILE):
            await event.reply("❌ No logs found yet.")
            return
        await event.reply("📤 Exporting system monitor logs...")
        await client.send_file(event.chat_id,
                               LOG_FILE,
                               caption="📊 SysMon Logs")

# Com/update.py
from telethon import events
import os
import hashlib
import json
import asyncio
import sys

__HELP__ = "!update → Restart the session\n!update log → Show changes in commands/files"
__CATEGORY__ = "Setup"
HASH_FILE = "mode/file_hashes.json"  # stores last file hashes


def setup(client, my_id):

    def compute_hash(file_path):
        """Compute SHA256 hash of a file"""
        h = hashlib.sha256()
        with open(file_path, "rb") as f:
            h.update(f.read())
        return h.hexdigest()

    def get_commands_in_file(file_path):
        """Return True if file defines __HELP__, else False"""
        try:
            with open(file_path, "r", encoding="utf-8") as f:
                code = f.read()
            return "__HELP__" in code
        except:
            return False

    @client.on(
        events.NewMessage(from_users=my_id, pattern=r"^!update(?:\s+log)?$"))
    async def update(event):
        try:
            command = event.text.strip().lower()

            # ---------------------------
            # !update log
            # ---------------------------
            if "log" in command:
                msg = await event.edit("📊 Gathering command changes...")
                await asyncio.sleep(0.5)

                current_hashes = {}
                file_changes = []

                # Scan all Com/*.py files that define __HELP__
                for filename in os.listdir("Com"):
                    if filename.endswith(".py"):
                        path = os.path.join("Com", filename)
                        if get_commands_in_file(path):
                            current_hashes[filename] = compute_hash(path)

                # Load previous hashes
                if os.path.exists(HASH_FILE):
                    with open(HASH_FILE, "r", encoding="utf-8") as f:
                        old_hashes = json.load(f)
                else:
                    old_hashes = {}

                # Detect changes
                for fname, hval in current_hashes.items():
                    if fname not in old_hashes:
                        file_changes.append(f"🟢 Added !{fname[:-3]}")
                    elif old_hashes[fname] != hval:
                        file_changes.append(f"✏️ Changed !{fname[:-3]}")

                for fname in old_hashes:
                    if fname not in current_hashes:
                        file_changes.append(f"❌ Deleted !{fname[:-3]}")

                if not file_changes:
                    file_changes.append("✅ No changes detected.")

                result_text = "📋 Command changes:\n" + "\n".join(file_changes)
                result_text += f"\n\n🔹 Total commands: {len(current_hashes)}"

                await msg.edit(result_text)

                # Update stored hashes
                os.makedirs(os.path.dirname(HASH_FILE), exist_ok=True)
                with open(HASH_FILE, "w", encoding="utf-8") as f:
                    json.dump(current_hashes, f, indent=2)

            # ---------------------------
            # !update (restart)
            # ---------------------------
            else:
                msg = await event.edit(
                    "🔄 Restarting the session to apply changes...")
                await asyncio.sleep(1)  # ensure message is edited
                os.execv(sys.executable, [sys.executable] + sys.argv)

        except Exception as e:
            await event.edit(f"❌ Error: {str(e)}")
